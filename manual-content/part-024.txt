      function minutes(shop) {
        if (state.traveller === "arrival")
          return +$("#arrivalStay").value || 45;
        const time = $("#boarding").value;
        if (!time) return "—";
        const [h, m] = time.split(":").map(Number),
          n = new Date(),
          t = new Date();
        t.setHours(h, m, 0, 0);
        if (t < n) t.setDate(t.getDate() + 1);
        const transit =
          shop.zone === state.zone
            ? 8
            : shop.zone === "central" || state.zone === "central"
              ? 16
              : 24;
        return Math.max(
          0,
          Math.round((t - n) / 60000) - +$("#buffer").value - transit,
        );
      }
      function score(s) {
        let x =
          (s.zone === state.zone ? 40 : 0) +
          (s.need.includes(state.need) ? 30 : 0);
        if (state.traveller === "arrival" && s.need.includes("arrival"))
          x += 50;
        if (state.traveller === "departure") {
          let m = minutes(s);
          if (typeof m === "number") x += Math.max(-30, Math.min(20, m));
        }
        return x;
      }

      function corridorRoute(fromZone, targetZone, target) {
        if (fromZone === targetZone) return [zones[fromZone].xy, target];
        const from = (
          corridorSpines[fromZone] || [zones.central.xy, zones[fromZone].xy]
        )
          .slice()
          .reverse();
        const to = corridorSpines[targetZone] || [
          zones.central.xy,
          zones[targetZone].xy,
        ];
        const points = [...from, ...to.slice(1), target];
        return points.filter(
          (point, index) =>
            index === 0 ||
            point[0] !== points[index - 1][0] ||
            point[1] !== points[index - 1][1],
        );
      }

      function qrMarkup(seed) {
        const size = 21;
        const finder = (x, y, fx, fy) => {
          const dx = x - fx,
            dy = y - fy;
          if (dx < 0 || dx > 6 || dy < 0 || dy > 6) return false;
          return (
            dx === 0 ||
            dx === 6 ||
            dy === 0 ||
            dy === 6 ||
            (dx >= 2 && dx <= 4 && dy >= 2 && dy <= 4)
          );
        };
        const hash = [...seed].reduce((sum, ch) => sum + ch.charCodeAt(0), 0);
        let html = "";
        for (let y = 0; y < size; y++)
          for (let x = 0; x < size; x++) {
            const on =
              finder(x, y, 0, 0) ||
              finder(x, y, 14, 0) ||
              finder(x, y, 0, 14) ||
              ((x * 17 + y * 11 + hash + x * y) % 5 < 2 && x > 7 && y > 7);
            html += `<i class="${on ? "on" : ""}"></i>`;
          }
        return html;
      }

          (b) => (b.onclick = () => renderPublic(b.dataset.publiczone)),
        );
        $$("[data-publicmap]").forEach(
          (b) =>
            (b.onclick = () => {
              let p = publicPoints.find((x) => x.id === b.dataset.publicmap);
              if (p) {
                state.zone = p.zone;
                state.mapFilter = "public";
                renderMap();
                $("#map").scrollIntoView({ behavior: "smooth" });
              }
            }),
        );
      }
      function renderAll() {
        renderReco();
        renderMap();
        renderCorridor(state.corridor || state.zone);
        renderPublic(state.publicZone || state.zone);
        renderJourney();
      }
      function init() {
        setOptions();
        let n = new Date(Date.now() + 90 * 60000);
        $("#boarding").value =
          String(n.getHours()).padStart(2, "0") +
          ":" +
          String(n.getMinutes()).padStart(2, "0");
        $$("[data-traveller]").forEach(
          (b) =>
            (b.onclick = () => {
              state.traveller = b.dataset.traveller;
              $$("[data-traveller]").forEach((x) =>
                x.classList.toggle("active", x === b),
              );
              updateHome();
            }),
        );
        $$("[data-entry]").forEach(
          (b) =>
            (b.onclick = () => {
              $$("[data-entry]").forEach((x) =>
                x.classList.toggle("active", x === b),
              );
              $("#scanPane").classList.toggle(
                "hidden",
                b.dataset.entry !== "scan",
              );
            }),
        );
        $$("[data-need]").forEach(
          (b) =>
            (b.onclick = () => {
              state.need = b.dataset.need;
              $$("[data-need]").forEach((x) =>
                x.classList.toggle("active", x === b),
              );
              renderAll();
            }),
        );

          "zone",
          "arrivalZone",
          "arrivalStay",
          "boarding",
          "gate",
          "buffer",
        ].forEach((id) =>
          $("#" + id).addEventListener("input", () => {
            if (id === "arrivalZone" && state.traveller === "arrival")
              state.zone = $("#arrivalZone").value;
            else if (id === "zone") state.zone = $("#zone").value;
            renderAll();
          }),
        );
        $("#showMap").onclick = () =>
          $("#map").scrollIntoView({ behavior: "smooth" });
        $("#adjust").onclick = () =>
          $(".control-panel").scrollIntoView({
            behavior: "smooth",
            block: "center",
          });
        $("#corridorMap").onclick = () =>
          $("#map").scrollIntoView({ behavior: "smooth" });
        $("#journeyMap").onclick = () =>
          $("#map").scrollIntoView({ behavior: "smooth" });
        $$("[data-plan-action]").forEach(
          (button) =>
            (button.onclick = () => {
              if (button.dataset.planAction === "buffer") {
                $("#recommend").scrollIntoView({ behavior: "smooth" });
                toast("可在行程匹配页调整安全缓冲时间。");
              } else {
                toast(
                  "收尾提示：保存文件，收妥电脑、证件与充电设备，并按现场导向返回登机口。",
                );
              }
            }),
        );
        $("#publicMap").onclick = () => {
          state.mapFilter = "public";
          renderMap();
          $("#map").scrollIntoView({ behavior: "smooth" });
        };
        $("#startTimer").onclick = () => {
          if (state.timer) return;
          state.timer = setInterval(() => {
            state.left = Math.max(0, state.left - 1);
            refreshTimer();
            if (!state.left) {
              clearInterval(state.timer);
              state.timer = null;
              $("#timerHint").textContent =
                "安全办公窗口已结束，请立即根据现场导向前往登机口。";
              toast("安全办公窗口已结束。");
            }
          }, 1000);
          $("#timerHint").textContent =
            "计时进行中，请同时关注航班显示、机场广播与现场通行情况。";
        };
        $("#resetTimer").onclick = () => {
          clearInterval(state.timer);
          state.timer = null;
          state.left = Math.max(
            0,
            (typeof minutes(currentShop()) === "number"
              ? minutes(currentShop())
              : 0) * 60,
          );
          refreshTimer();
          $("#timerHint").textContent = "已按当前安全办公窗口重置计时。";
        };
        $("#copyPlan").onclick = async () => {
          let t = $("#journeyCopy").textContent;
          try {
            await navigator.clipboard.writeText(t);
            toast("安全办公计划已复制。");
          } catch (e) {
            toast(t);
          }
        };
        $("#camera").onclick = () => $("#boardingFile").click();
        $("#upload").onclick = () => $("#boardingFile").click();

          openAvailability(currentShop().id);
        $("#homeCapacity").onclick = () => openAvailability(currentShop().id);

          if (e.target.files[0]) {
            $("#scanStatus").textContent =
              "登机牌已载入。请人工核对登机时间与登机口后继续；航班状态将在授权后由航旅纵横校验。";
            toast("已载入登机牌，请完成信息核对。");
          }
        };
        $("#closeAvailability").onclick = () =>
          $("#availabilityModal").classList.add("hidden");
        $("#closeReservation").onclick = () =>
          $("#reservationModal").classList.add("hidden");
        $("#backToAvailability").onclick = () => {
          $("#reservationModal").classList.add("hidden");
          $("#availabilityModal").classList.remove("hidden");
        };
        $("#getReservationQr").onclick = () => {
          const phone = $("#reservationPhone").value.trim();
          if (!/^1\d{10}$/.test(phone)) {
            toast("请输入有效的11位手机号码。");
            return;
          }
          $("#qrGrid").innerHTML = qrMarkup(phone + currentShop().id);
          $("#qrTitle").textContent = `${currentShop().name}｜预约凭证`;
          $("#qrPlaceholder").classList.add("hidden");
          $("#qrResult").classList.remove("hidden");
          $("#completeReservation").classList.remove("hidden");
        };
        $("#completeReservation").onclick = () => {
          $("#reservationModal").classList.add("hidden");
          toast("预约凭证已生成，请向门店服务人员出示。");
        };
        $("#closePhoto").onclick = () =>
          $("#photoModal").classList.add("hidden");

          $("#availabilityModal").classList.add("hidden");
          $("#recommend").scrollIntoView({ behavior: "smooth" });

        $("#reserve").onclick = () => {
          $("#availabilityModal").classList.add("hidden");
          openReservation();
        };
        $("#print").onclick = () => window.print();
        $("#reset").onclick = () => location.reload();
        const obs = new IntersectionObserver(
          (es) =>
            es.forEach((e) => {
              if (e.isIntersecting)
                $$(".bottom a").forEach((a) =>
                  a.classList.toggle(
                    "active",
                    a.getAttribute("href") === "#" + e.target.id,
                  ),
                );
            }),
          { threshold: 0.45 },
        );
        $$(".page").forEach((p) => obs.observe(p));
        renderAll();
      }
      init();
